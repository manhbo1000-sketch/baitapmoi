        <%
            try {
                // Lấy dữ liệu từ Request và ép kiểu
                String rateStr = request.getParameter("rate");
                String usdStr  = request.getParameter("usd");

                // Kiểm tra null/rỗng
                if (rateStr == null || usdStr == null
                        || rateStr.trim().isEmpty() || usdStr.trim().isEmpty()) {
                    throw new IllegalArgumentException("Thiếu tham số đầu vào");
                }

                float rate = Float.parseFloat(rateStr);
                float usd  = Float.parseFloat(usdStr);

                // Kiểm tra giá trị hợp lệ
                if (rate <= 0 || usd < 0) {
                    throw new IllegalArgumentException("Giá trị không hợp lệ");
                }

                // Tính toán: VND = USD * Rate
                float vnd = rate * usd;

                // Format số theo kiểu Việt Nam (dấu chấm phân cách nghìn)
                NumberFormat nf = NumberFormat.getInstance(new Locale("vi", "VN"));
                nf.setMaximumFractionDigits(2);

                String rateFmt = nf.format(rate);
                String usdFmt  = nf.format(usd);
                String vndFmt  = nf.format(vnd);
        %>

        <%-- ====================================================
             EXPRESSION: Hiển thị dữ liệu ra HTML
             ==================================================== --%>
        <div class="info-row">
            Tỉ giá hiện tại: <b><%= rateFmt %></b> VND/USD
        </div>
        <div class="info-row">
            Lượng USD yêu cầu: <b>$<%= usdFmt %></b>
        </div>

        <div class="result-box">
            <h3>💰 Thành tiền: <%= vndFmt %> VNĐ</h3>
        </div>

        <%
            // Kết thúc khối try
            } catch (NumberFormatException e) {
        %>
        <div class="error-box">

            <p>Dữ liệu nhập vào không phải là số. Vui lòng kiểm tra lại.</p>
        </div>
        <%
            } catch (Exception e) {
        %>
        <div class="error-box">
            <h3>⚠️ Lỗi xử lý!</h3>

        </div>
        <%
            }
